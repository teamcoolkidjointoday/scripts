-- Utility Functions to Detect Roles
local function getPlayerRole(player)
    if not player or not player:IsA("Player") then return "Innocent" end
    
    local character = player.Character
    local backpack = player.Backpack
    
    local hasKnife = false
    local hasGun = false
    
    if backpack then
        if backpack:FindFirstChild("Knife") then hasKnife = true end
        if backpack:FindFirstChild("Gun") then hasGun = true end
    end
    
    if character then
        if character:FindFirstChild("Knife") then hasKnife = true end
        if character:FindFirstChild("Gun") then hasGun = true end
    end
    
    if hasKnife then
        return "Murderer"
    elseif hasGun then
        return "Sheriff"
    else
        return "Innocent"
    end
end

local function applyESP(player)
    if player == LocalPlayer then return end
    
    local function setupHighlight(character)
        if not ESP_Enabled then return end
        if Highlights[player] then Highlights[player]:Destroy() end
        
        local Highlight = Instance.new("Highlight")
        Highlight.Parent = character
        Highlight.Adornee = character
        Highlight.FillTransparency = 0.5
        Highlight.OutlineTransparency = 0.1
        
        Highlights[player] = Highlight
        
        -- Dynamic Role Detection Loop
        task.spawn(function()
            while character and character.Parent and ESP_Enabled do
                local role = getPlayerRole(player)
                if role == "Murderer" then
                    Highlight.FillColor = Color3.fromRGB(255, 0, 0)
                    Highlight.OutlineColor = Color3.fromRGB(255, 50, 50)
                elseif role == "Sheriff" then
                    Highlight.FillColor = Color3.fromRGB(0, 0, 255)
                    Highlight.OutlineColor = Color3.fromRGB(50, 50, 255)
                else
                    Highlight.FillColor = Color3.fromRGB(0, 255, 0)
                    Highlight.OutlineColor = Color3.fromRGB(50, 255, 50)
                end
                task.wait(1)
            end
        end)
    end
    
    if player.Character then
        setupHighlight(player.Character)
    end
    
    player.CharacterAdded:Connect(function(char)
        if ESP_Enabled then
            task.wait(0.5)
            setupHighlight(char)
        end
    end)
end

local function removeESP()
    for player, highlight in pairs(Highlights) do
        if highlight then
            highlight:Destroy()
        end
    end
    table.clear(Highlights)
end

createToggle(ESP_Page, "ESP (Innocent, Sheriff, Murderer)", false, function(state)
    ESP_Enabled = state
    if ESP_Enabled then
        for _, player in ipairs(Players:GetPlayers()) do
            applyESP(player)
        end
        Players.PlayerAdded:Connect(applyESP)
    else
        removeESP()
    end
end)

local function findDroppedGun()
    local targetGun = nil
    
    -- Search for "GunDrop" model
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj.Name == "GunDrop" then
            targetGun = obj
            break
        end
    end
    
    -- Fallback search for unheld tool
    if not targetGun then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Tool") and (obj.Name == "Gun" or obj.Name == "Luger") then
                local isHeld = obj:FindFirstAncestorOfClass("Model") and obj:FindFirstAncestorOfClass("Model"):FindFirstChildOfClass("Humanoid")
                if not isHeld then
                    targetGun = obj
                    break
                end
            end
        end
    end
    
    return targetGun
end

local function grabGunTeleport()
    local character = LocalPlayer.Character
    if not character then return end
    
    local targetGun = findDroppedGun()
    if targetGun then
        if targetGun:IsA("Model") then
            local targetPart = targetGun.PrimaryPart or targetGun:FindFirstChildWhichIsA("BasePart")
            if targetPart then
                character:PivotTo(targetPart.CFrame * CFrame.new(0, 1, 0))
            else
                character:PivotTo(targetGun:GetPivot() * CFrame.new(0, 1, 0))
            end
        elseif targetGun:IsA("BasePart") or targetGun:IsA("Tool") then
            character:PivotTo(targetGun:GetPivot() * CFrame.new(0, 1, 0))
        end
    end
end

createButton(Innocent_Page, "Grab Gun (Teleport To Gun)", function()
    grabGunTeleport()
end)

local function getKnife()
    local character = LocalPlayer.Character
    if not character then return nil end
    local knife = character:FindFirstChild("Knife") or LocalPlayer.Backpack:FindFirstChild("Knife")
    if knife then
        knife.Parent = character
        return knife
    end
    return nil
end

local function killTargetRemote(targetPlayer)
    local character = LocalPlayer.Character
    local targetChar = targetPlayer.Character
    
    if character and targetChar then
        local myHrp = character:FindFirstChild("HumanoidRootPart")
        local targetHrp = targetChar:FindFirstChild("HumanoidRootPart")
        local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
        local knife = getKnife()
        
        if myHrp and targetHrp and targetHum and targetHum.Health > 0 and knife then
            local knifeHandle = knife:FindFirstChild("Handle") or knife:FindFirstChildWhichIsA("BasePart")
            if knifeHandle then
                knife:Activate()
                
                if firetouchinterest then
                    firetouchinterest(targetHrp, knifeHandle, 0)
                    task.wait()
                    firetouchinterest(targetHrp, knifeHandle, 1)
                else
                    local previousCFrame = myHrp.CFrame
                    myHrp.CFrame = targetHrp.CFrame * CFrame.new(0, 0, 1)
                    task.wait(0.02)
                    knife:Activate()
                    myHrp.CFrame = previousCFrame
                end
            end
        end
    end
end

-- Custom Dragging System Implementation
local dragging
local dragInput
local dragStart
local startPos

local function update(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

game:GetService("UserInputService").InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)
